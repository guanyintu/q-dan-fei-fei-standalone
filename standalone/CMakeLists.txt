cmake_minimum_required(VERSION 3.24)
project(q弹狒狒 VERSION 0.2.2 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options($<$<COMPILE_LANGUAGE:CXX>:/utf-8> $<$<COMPILE_LANGUAGE:CXX>:/W4>)
add_subdirectory(vendor/minhook EXCLUDE_FROM_ALL)
add_library(QBounceModule SHARED module.cpp)
target_link_libraries(QBounceModule PRIVATE minhook user32)
set_target_properties(QBounceModule PROPERTIES OUTPUT_NAME "QBounceModule")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/embedded.rc" CONTENT
"#pragma code_page(65001)\n1 24 \"${CMAKE_CURRENT_SOURCE_DIR}/app.manifest\"\n101 RCDATA \"$<TARGET_FILE:QBounceModule>\"\n102 RCDATA \"${CMAKE_CURRENT_SOURCE_DIR}/../QBounce/Assets/bgm.mp3\"\n1 VERSIONINFO\nFILEVERSION 0,2,2,0\nPRODUCTVERSION 0,2,2,0\nFILEOS 0x40004\nFILETYPE 1\nBEGIN\n BLOCK \"StringFileInfo\"\n BEGIN\n  BLOCK \"080404B0\"\n  BEGIN\n   VALUE \"CompanyName\", \"someboy\\0\"\n   VALUE \"FileDescription\", \"会让所有人变得q弹\\0\"\n   VALUE \"FileVersion\", \"${PROJECT_VERSION}\\0\"\n   VALUE \"InternalName\", \"q弹狒狒\\0\"\n   VALUE \"OriginalFilename\", \"q弹狒狒.exe\\0\"\n   VALUE \"ProductName\", \"q弹狒狒\\0\"\n   VALUE \"ProductVersion\", \"${PROJECT_VERSION}\\0\"\n   VALUE \"LegalCopyright\", \"someboy\\0\"\n  END\n END\n BLOCK \"VarFileInfo\"\n BEGIN\n  VALUE \"Translation\", 0x0804, 1200\n END\nEND\n")
add_executable(QBounceStandalone WIN32 launcher.cpp "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/embedded.rc")
add_dependencies(QBounceStandalone QBounceModule)
set_target_properties(QBounceStandalone PROPERTIES OUTPUT_NAME "q弹狒狒-${PROJECT_VERSION}")
target_link_options(QBounceStandalone PRIVATE /MANIFEST:NO)
target_link_libraries(QBounceStandalone PRIVATE comctl32 comdlg32 shell32 shlwapi psapi mfplat mfuuid ole32 oleaut32 gdiplus dwmapi uxtheme)
add_executable(QBounceNativeTests tests.cpp)
add_executable(QBouncePerfBench benchmark.cpp)
target_link_libraries(QBouncePerfBench PRIVATE minhook user32)
enable_testing()
add_test(NAME NativeCore COMMAND QBounceNativeTests)
add_test(NAME NativeFrameWork COMMAND QBouncePerfBench --assert-efficient)
